\begin{algorithm}[t]
\caption{Sampling valid time series}
\label{alg:sample-children-tree}
\begin{algorithmic}[1]
\Require Simulator $\mathcal{S}$; intervenable parameter set $\Theta^\star$;
         initial-state prior $p(x_0)$;
         intervention-time prior $p(t_{\mathrm{int}})$;

\State $\mathcal{C} \gets \Theta^\star \cup \{\textsc{NoChange}\}$ \Comment{Candidate labels}
\State $\mathcal{Y} \gets \emptyset$

\While{$|\mathcal{Y}| < \mathrm{target}$}
    \State Sample baseline parameters $\Theta^\star \sim p(\Theta^\star)$ and initial state $x_0 \sim p(x_0)$
    \State Simulate the parent trajectory
    \[
        y_{\mathrm{parent}}(t) = \mathcal{S}(\theta, x_0).
    \]

    \State Initialize children set $\mathcal{T} \gets \emptyset$

    \ForAll{$c \in \mathcal{C}$}
        \If{$c = \textsc{NoChange}$}
            \State $y_c(t) \gets y_{\mathrm{parent}}(t)$

            \State Add child $(\tilde{y}_c, c)$ to $\mathcal{T}$
        \Else
            \State Sample intervention time $t_{\mathrm{int}} \sim p(t_{\mathrm{int}})$
            \State Sample an intervened value $\theta'_c$ for parameter $c$
            \State Construct post-intervention parameters
            \[
                \theta^{(c)}_{\mathrm{post}}
                \gets
                \theta
                \quad\text{with}\quad
                \theta_c \leftarrow \theta'_c .
            \]
            \State Simulate the pre-intervention segment until $t_{\mathrm{int}}$
            \[
                y_{\mathrm{pre}}(t) = \mathcal{S}(\theta, x_0),
                \qquad t < t_{\mathrm{int}} .
            \]
            \State Simulate the corresponding countertefactual baseline with the intervened parameter from the beginning
               \[
                y_{\mathrm{time0parent}}(t) = \mathcal{S}(\theta^{(c)}_{\mathrm{post}}, x_0).
            \]

            \State Let $x_{\mathrm{int}}$ be the simulator state reached at $t_{\mathrm{int}}$
            \State Simulate the post-intervention segment from the same physical state
            \[
                y^{(c)}_{\mathrm{post}}(\tau)
                =
                \mathcal{S}(\theta^{(c)}_{\mathrm{post}}, x_{\mathrm{int}}),
                \qquad \tau \geq 0 .
            \]
            \State Concatenate the two segments:
            \[
                y_c(t)
                =
                \begin{cases}
                y_{\mathrm{pre}}(t), & t < t_{\mathrm{int}},\\
                y^{(c)}_{\mathrm{post}}(t - t_{\mathrm{int}}), & t \geq t_{\mathrm{int}} .
                \end{cases}
            \]

            \State Perform detectability checks:
            \begin{enumerate}
                \item $y_c(t) \neq y_{\mathrm{parent}}(t)$
                \item $y_c(t) \neq y_{\mathrm{time0parent}}(t)$
                \item Environment-specific checks pass.
            \end{enumerate}
            \If{all detectability checks pass}
                \State Add child $(\tilde{y}_c, c)$ to $\mathcal{T}$
            \EndIf
        \EndIf

    \EndFor
    \State Sample one or more children from $\mathcal{T}$ and add them to $\mathcal{Y}_{\mathrm{test}}$
\EndWhile

\end{algorithmic}
\end{algorithm}
